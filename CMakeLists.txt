cmake_minimum_required(VERSION 3.20)
project(Arcade VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(arcade_engine src/library.cpp src/core.cpp src/cover_art.cpp)
target_include_directories(arcade_engine PUBLIC src vendor)
target_link_libraries(arcade_engine PUBLIC ${CMAKE_DL_LIBS})
target_compile_options(arcade_engine PRIVATE -Wall -Wextra -Wpedantic)
add_library(arcade_model src/settings.cpp src/settings_menu.cpp src/model.cpp)
target_include_directories(arcade_model PUBLIC src)
target_link_libraries(arcade_model PUBLIC arcade_engine)
target_compile_options(arcade_model PRIVATE -Wall -Wextra -Wpedantic)

# Production UI is custom C++/OpenGL, with native platform adapters.
if(APPLE)
 find_package(CURL REQUIRED)
 enable_language(OBJCXX)
 add_executable(arcade native/main.cpp native/gfx/renderer.cpp native/ui/gallery.cpp native/ui/vault.cpp native/platform/macos.mm native/platform/cover_http.cpp)
 target_include_directories(arcade PRIVATE native)
 target_link_libraries(arcade PRIVATE arcade_model CURL::libcurl "-framework Cocoa" "-framework OpenGL" "-framework AudioToolbox")
 target_compile_definitions(arcade PRIVATE ARCADE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_options(arcade PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-function)
endif()
